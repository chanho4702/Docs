    static int resolvePeriodSeconds(Instant from, Instant to, Instant now) {
        int base = from.isAfter(now.minus(Duration.ofDays(63))) ? 300 : 3600;
        long rangeSeconds = Math.max(1, Duration.between(from, to).getSeconds());
        long minPeriod = (long) Math.ceil(rangeSeconds / (double) MAX_DATAPOINTS);
        long multiples = Math.max(1, (long) Math.ceil(minPeriod / (double) base));
        return (int) (multiples * base);
    }

    private static boolean isAccessDenied(AwsServiceException e) {

        return e.statusCode() == 403

                || "UnauthorizedOperation".equals(code);

    private static double statisticValue(Datapoint p, Statistic statistic) {
        Double value = statistic == Statistic.AVERAGE ? p.average()
                : statistic == Statistic.MAXIMUM ? p.maximum()
                        : statistic == Statistic.MINIMUM ? p.minimum() : p.sum();
        return value == null ? 0d : value;
    }

    private boolean reverseUnique(AgentHost host, String instanceId) {
        Map<String, Set<String>> index = ec2AddressIndex.get();
        if (index.isEmpty()) {
            return false;

        Set<String> owners = new HashSet<>();
        for (String address : host.addresses()) {
            Set<String> ids = index.get(address);
            if (ids != null) {
                owners.addAll(ids);
            }
        }
        return owners.size() == 1 && owners.contains(instanceId);
    }

    static Set<String> instanceAddresses(Instance instance) {
        Set<String> addresses = new LinkedHashSet<>();
        addIfPresent(addresses, instance.privateIpAddress());
        addIfPresent(addresses, instance.publicIpAddress());
        for (var eni : instance.networkInterfaces()) {
            for (var ip : eni.privateIpAddresses()) {
                addIfPresent(addresses, ip.privateIpAddress());
                if (ip.association() != null) {
                    addIfPresent(addresses, ip.association().publicIp());
                }
            }
        }
        return addresses;
    }

    private static void addIfPresent(Set<String> into, String address) {
        if (address != null && !address.isBlank()) {
            into.add(address.trim());
        }
    }

    private synchronized Snapshot hosts() {

        if (current != null && current.fetchedAt().isAfter(Instant.now().minus(CACHE_TTL))) {
            return current;
        }

                    new HostQuery(null, null, Set.of(HostInclude.TAGS, HostInclude.LAST_CHECK)));

                    if (iface.kind() == InterfaceKind.AGENT && !Boolean.FALSE.equals(iface.useIp())) {

    private RegionSnapshot collectRegion(LocalDateTime runAt, String region) {
        List<Instance> instances = aws.describeInstances(region);

        Map<String, String> statuses = new LinkedHashMap<>();
        List<String> ids = instances.stream().map(Instance::instanceId).toList();
        if (!ids.isEmpty()) {

        List<CloudInstanceSnapshot> instanceRows = new ArrayList<>();
        for (Instance i : instances) {

            instanceRows.add(CloudInstanceSnapshot.builder()
                    .runAt(runAt).region(region)
                    .instanceId(i.instanceId()).name(nameTag(i.tags()))
                    .state(state).statusCheck(check)
                    .severity(Ec2StatusRules.classify(state, check))
                    .build());
        }

        List<CloudVpcSnapshot> vpcRows = new ArrayList<>();
        for (Vpc vpc : aws.describeVpcs(region)) {
            List<Subnet> subnets = aws.describeSubnetsByVpc(region, vpc.vpcId());
            List<RouteTable> routeTables = aws.describeRouteTablesByVpc(region, vpc.vpcId());
            List<SecurityGroup> groups = aws.describeSecurityGroupsByVpc(region, vpc.vpcId());
            List<Map<String, Object>> findings = NetworkExposureScanner.findings(subnets, routeTables, groups);
            long critical = findings.stream().filter(f -> "critical".equals(f.get("severity"))).count();
            // ?몄뒪?댁뒪 ?뚯냽? ?대? 諛쏆븘 ??由ъ쟾 紐⑸줉?먯꽌 ?몃㈃ ?쒕떎 ??describeInstancesByVpc ?ы샇異쒖? ??퉬.
            long inVpc = instances.stream().filter(i -> vpc.vpcId().equals(i.vpcId())).count();
            vpcRows.add(CloudVpcSnapshot.builder()
                    .runAt(runAt).region(region)
                    .vpcId(vpc.vpcId()).name(nameTag(vpc.tags()))

    private String nameTag(List<Tag> tags) {

    }

    /** 由ъ쟾蹂?蹂묐젹 ?섏쭛. ?ㅽ뙣??由ъ쟾? null ???몄텧遺媛 failed濡??쇰떎. */
    private <T> List<T> runParallel(List<? extends Callable<T>> tasks) {
        if (tasks.isEmpty()) {
            return List.of();
        }
        ExecutorService pool = Executors.newFixedThreadPool(Math.min(tasks.size(), MAX_PARALLELISM));
        try {
            List<Future<T>> futures = tasks.stream().map(pool::submit).toList();
            List<T> results = new ArrayList<>(futures.size());
            for (Future<T> future : futures) {
                try {
                    results.add(future.get());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    results.add(null);
                } catch (Exception e) {
                    log.warn("cloud snapshot region task failed: {}", e.getMessage());
                    results.add(null);
                }
            }
            return results;
        } finally {
            pool.shutdown();
        }
    }

            List<Instance> rawInstances) {
    }

/**
 * "이 EC2가 어느 Zabbix 호스트인가" — 소스 매트릭스의 매칭 축
 * (devdesign_ec2-monitoring-sources.md §3, 코덱스 검토 §8 반영).
 *
 * <p>매칭 규칙(자동 연결은 보수적으로 — <b>오연결이 미연결보다 나쁘다</b>):
 * <ul>
 *   <li>1순위 {@code INSTANCE_ID_EXACT}: Zabbix 호스트명 == instanceId, 또는 호스트 태그
 *       {@code aws.instanceId} == instanceId. 명시 식별자라 가장 안전.</li>
 *   <li>2순위 {@code IP_EXACT}: EC2의 <b>ENI 전체 주소</b>(secondary IP·EIP 포함) ↔ Zabbix
 *       <b>AGENT 인터페이스</b> 주소. 인스턴스가 정확히 한 호스트만 때릴 때만 연결
 *       (사설 IP는 VPC 간 중복 가능).</li>
 *   <li>Name 태그 == 호스트명 매칭은 하지 않는다 — 이름은 중복·변경이 흔하다(코덱스 지적).</li>
 * </ul>
 *
 * <p>🔴 Zabbix 조회는 <b>시스템 컨텍스트</b>({@link CallerContext#system()}) + 전역 5분 캐시다 —
 * 사용자별 세션 결과를 전역 캐시에 섞으면 권한 누출이 된다(코덱스가 잡은 리스크). 이 매칭 결과는
 * awsResources.view 권한으로 게이트된 클라우드 화면에서만 소비된다.
 *
 * <p>⚠️ 알려진 한계: useip=0(DNS 접속) Zabbix 호스트는 InventoryPort 모델이 주소만 주므로 매칭 누락될
 * 수 있다. 수동 연결(오버라이드)은 자산 축과 함께 2차.
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class Ec2ZabbixMatcher {

    public static final String BASIS_INSTANCE_ID = "INSTANCE_ID_EXACT";
    public static final String BASIS_IP = "IP_EXACT";
    /** Zabbix 호스트 태그로 명시 연결하는 키(권장 온보딩 — 문서화 대상). */
    static final String TAG_INSTANCE_ID = "aws.instanceId";

    private static final Duration CACHE_TTL = Duration.ofMinutes(5);
    private static final SourceRef SRC = SourceRef.DEFAULT;

    private final InventoryPort inventoryPort;
    private final AtomicReference<Snapshot> cache = new AtomicReference<>();
    /**
     * EC2 주소 → instanceId 역색인(전 리전). IP_EXACT의 <b>역방향 유일성</b> 검사 재료 —
     * "이 호스트의 주소를 쓰는 EC2가 정확히 이 인스턴스 하나"여야 연결한다(코덱스 2차: 머지 차단 사유).
     * 🔴 스냅샷 배치(5분, 전 리전)가 채운다. 비어 있으면(콜드) IP 근거는 쓰지 않는다 — fail-closed.
     */
    private final AtomicReference<Map<String, Set<String>>> ec2AddressIndex = new AtomicReference<>(Map.of());

    /** 매칭 결과 — basis·판정 시각을 보존한다("왜 연결됐나"에 화면이 답할 수 있게). */
    public record Match(String hostid, String hostName, String basis, Long lastCheckEpoch, Instant matchedAt) {
    }

    /**
     * agent 수집이 신선한가 — 커버리지 판정의 freshness 프록시(LAST_CHECK 기반, MetricPort 호출 없음).
     */
    public boolean isFresh(Match match, Duration tolerance) {
        return match.lastCheckEpoch() != null
                && Instant.ofEpochSecond(match.lastCheckEpoch()).isAfter(Instant.now().minus(tolerance));
    }

    /**
     * EC2 주소 역색인 갱신 — 스냅샷 배치가 전 리전 인스턴스를 넘긴다(통째 교체, 삭제된 인스턴스 자연 소거).
     */
    public void indexInstances(List<Instance> instances) {
        Map<String, Set<String>> index = new HashMap<>();
        for (Instance instance : instances) {
            for (String address : instanceAddresses(instance)) {
                index.computeIfAbsent(address, k -> new HashSet<>()).add(instance.instanceId());
            }
        }
        ec2AddressIndex.set(index);
    }

    /**
     * 단일 인스턴스 매칭. IP 축은 양방향 유일성을 요구한다:
     * ① 인스턴스 주소들이 정확히 한 AGENT 호스트만 때리고 ② 그 호스트의 주소들을 쓰는 EC2가 이 인스턴스뿐.
     * ②는 {@link #indexInstances} 역색인으로 판정하며, 색인이 비어 있으면(콜드) IP 근거를 쓰지 않는다.
     */
    public Optional<Match> matchFor(Instance instance) {
        Snapshot snapshot = hosts();
        if (snapshot == null || instance == null) {
            return Optional.empty();
        }
        // 1순위: 명시 식별자
        AgentHost byId = snapshot.byInstanceId().get(instance.instanceId());
        if (byId != null) {
            return Optional.of(match(byId, BASIS_INSTANCE_ID));
        }
        // 2순위: IP — ENI 전체 주소로 유일 호스트만
        Set<AgentHost> hits = new LinkedHashSet<>();
        for (String address : instanceAddresses(instance)) {
            List<AgentHost> candidates = snapshot.byAddress().get(address);
            if (candidates != null) {
                hits.addAll(candidates);
            }
        }
        if (hits.size() == 1) {
            AgentHost host = hits.iterator().next();
            if (reverseUnique(host, instance.instanceId())) {
                return Optional.of(match(host, BASIS_IP));
            }
            log.debug("ec2 {} ↔ host {}: ip shared by other instances or index cold — not linking",
                    instance.instanceId(), host.hostid());
            return Optional.empty();
        }
        if (hits.size() > 1) {
            log.debug("ec2 {} matches {} zabbix hosts by ip — not linking", instance.instanceId(), hits.size());
        }
        return Optional.empty();
    }

    /**
     * 역방향 유일성 — 호스트 주소를 쓰는 EC2가 이 인스턴스 하나뿐인가. 다른 VPC의 EC2가 같은 사설 IP를
     * 쓰면 둘 다 같은 호스트 지표를 받게 되므로(코덱스 2차) 그 경우엔 연결하지 않는다.
     * 색인이 비었거나(콜드) 호스트 주소가 색인에 없으면 판정 불가 → false(fail-closed).
     */
    private boolean reverseUnique(AgentHost host, String instanceId) {
        Map<String, Set<String>> index = ec2AddressIndex.get();
        if (index.isEmpty()) {
            return false;
        }
        Set<String> owners = new HashSet<>();
        for (String address : host.addresses()) {
            Set<String> ids = index.get(address);
            if (ids != null) {
                owners.addAll(ids);
            }
        }
        return owners.size() == 1 && owners.contains(instanceId);
    }

    /** EC2의 매칭 후보 주소 전체 — primary 외 ENI의 secondary private IP·연결 public IP 포함(코덱스 지적). */
    static Set<String> instanceAddresses(Instance instance) {
        Set<String> addresses = new LinkedHashSet<>();
        addIfPresent(addresses, instance.privateIpAddress());
        addIfPresent(addresses, instance.publicIpAddress());
        for (var eni : instance.networkInterfaces()) {
            for (var ip : eni.privateIpAddresses()) {
                addIfPresent(addresses, ip.privateIpAddress());
                if (ip.association() != null) {
                    addIfPresent(addresses, ip.association().publicIp());
                }
            }
        }
        return addresses;
    }

    private static void addIfPresent(Set<String> into, String address) {
        if (address != null && !address.isBlank()) {
            into.add(address.trim());
        }
    }

    private Match match(AgentHost host, String basis) {
        return new Match(host.hostid(), host.name(), basis, host.lastCheckEpoch(), Instant.now());
    }

    // ===== Zabbix 호스트 캐시 =====

    record AgentHost(String hostid, String name, Set<String> addresses, Long lastCheckEpoch) {
    }

    private record Snapshot(Instant fetchedAt, Map<String, List<AgentHost>> byAddress,
            Map<String, AgentHost> byInstanceId) {
    }

    /**
     * 캐시 로딩 — synchronized로 single-flight: 개요의 리전 병렬 요청이 콜드 캐시에서 동시에 Zabbix 전체
     * 호스트를 N번 조회하는 것을 막는다(코덱스 §7). TTL 內 경로는 잠금 비용만 치른다.
     */
    private synchronized Snapshot hosts() {
        Snapshot current = cache.get();
        if (current != null && current.fetchedAt().isAfter(Instant.now().minus(CACHE_TTL))) {
            return current;
        }
        try {
            List<InventoryHost> hosts = inventoryPort.listHosts(SRC, CallerContext.system(),
                    new HostQuery(null, null, Set.of(HostInclude.TAGS, HostInclude.LAST_CHECK)));
            Map<String, List<AgentHost>> byAddress = new HashMap<>();
            Map<String, AgentHost> byInstanceId = new HashMap<>();
            Set<String> ambiguousInstanceIds = new HashSet<>();
            for (InventoryHost host : hosts) {
                Set<String> addresses = new LinkedHashSet<>();
                for (HostNetworkInterface iface : host.interfaces()) {
                    // 🔴 AGENT 인터페이스만 — SNMP/IPMI 주소로 "agent 있음" 판정하면 성능 지표가 없는데
                    //    있다고 말하는 셈이 된다(코덱스 지적).
                    // 🔴 useip=0(DNS 접속) 인터페이스는 제외 — 그 호스트의 ip 필드는 접속에 안 쓰는 설정값이라
                    //    EC2 주소와 우연히 같아도 근거가 아니다(코덱스 검수 §2). useIp 미상(null)은 포함.
                    if (iface.kind() == InterfaceKind.AGENT && !Boolean.FALSE.equals(iface.useIp())) {
                        addIfPresent(addresses, iface.address());
                    }
                }
                if (addresses.isEmpty()) {
                    continue;
                }
                AgentHost agentHost = new AgentHost(host.ref().nativeId(), host.name(), addresses, host.lastCheckEpoch());
                for (String address : addresses) {
                    byAddress.computeIfAbsent(address, k -> new java.util.ArrayList<>()).add(agentHost);
                }
                for (String instanceId : instanceIdClaims(host)) {
                    // 같은 instanceId를 두 호스트가 주장하면 둘 다 버린다 — 유일성 규칙.
                    if (byInstanceId.put(instanceId, agentHost) != null) {

    private static Set<String> instanceIdClaims(InventoryHost host) {
        Set<String> claims = new LinkedHashSet<>();
        if (host.name() != null && host.name().startsWith("i-")) {
            claims.add(host.name().trim());
        }
        host.tags().stream()
                .filter(tag -> TAG_INSTANCE_ID.equals(tag.name()))
                .map(tag -> tag.value())
                .filter(v -> v != null && !v.isBlank())
                .forEach(v -> claims.add(v.trim()));
        return claims;
    }
