    public Map<String, Object> metrics(String instanceId, String region, long fromEpoch, long toEpoch) {
        Instant now = Instant.now();
        Instant to = toEpoch > 0 ? Instant.ofEpochSecond(toEpoch) : now;
        Instant from = fromEpoch > 0 ? Instant.ofEpochSecond(fromEpoch) : to.minus(Duration.ofDays(1));

        Instant denied = deniedUntil.get();
        if (denied != null && now.isBefore(denied)) {
            return degraded("unavailable", "CLOUDWATCH_ACCESS_DENIED");
        }
        String month = LocalDate.now().format(DateTimeFormatter.ofPattern("yyyy-MM"));
        long used = budgetRepository.findById(month).map(b -> b.getCalls()).orElse(0L);
        if (used + METRICS.size() > monthlyCallCap) {
            log.warn("cloud metrics budget exhausted: used={} cap={}", used, monthlyCallCap);
            return degraded("disabled", "BUDGET_EXHAUSTED");

        Map<String, Object> series = new LinkedHashMap<>();

                series.put(metric.key(), points.stream()
                        // 臾몄꽌 紐낆떆: ?곗씠?고룷?명듃???쒓컙?쒖씠 ?꾨땲?????ш린???뺣젹??FE 遺?댁쓣 ?놁븻??
                        .sorted(Comparator.comparing(Datapoint::timestamp))

    static int resolvePeriodSeconds(Instant from, Instant to, Instant now) {

        long rangeSeconds = Math.max(1, Duration.between(from, to).getSeconds());
        long minPeriod = (long) Math.ceil(rangeSeconds / (double) MAX_DATAPOINTS);
        long multiples = Math.max(1, (long) Math.ceil(minPeriod / (double) base));
        return (int) (multiples * base);

    /** overview(16)蹂대떎 蹂댁닔?곸쑝濡???諛곗튂??吏?곗씠 湲됲븯吏 ?딄퀬, 怨좉컼 怨꾩젙??API rate瑜??꾨??? */
    private static final int MAX_PARALLELISM = 8;

    private final AwsReadOnlyGateway aws;
    private final CloudCredentialsResolver credentials;
    private final CloudInstanceSnapshotRepository instanceRepository;
    private final CloudVpcSnapshotRepository vpcRepository;

    /**
     * ?섏쭛 1?? 由ъ쟾 ?ъ븘?????몄뒪?댁뒪쨌VPC ?ㅻ깄???곸옱.
     *
     * @return ?ㅽ뻾 濡쒓렇 ?붿빟. ?먭꺽利앸챸???놁쑝硫?<b>?덉쇅媛 ?꾨땲??skip</b> ???대씪?곕뱶瑜????곕뒗
     *     ?ㅼ튂蹂몄뿉????諛곗튂媛 留?5遺?"?ㅽ뙣"濡?李랁엳硫????쒕떎(?뺤긽 ?곹깭??.
     */
    public String collect() {
        if (credentials.source() == CloudCredentialsResolver.CredentialSource.NONE) {
            return "skipped: unconfigured";
        }
        LocalDateTime runAt = LocalDateTime.now();
        // ?뵶 ?쒖꽦 由ъ쟾留???allRegions??誘몄샃?몄씤 由ъ쟾源뚯? 以??ъ븘???덈컲???몄쬆 ?ㅽ뙣濡??쏅룉???ㅼ륫).
        List<String> regions = aws.enabledRegionNames();

        List<Callable<RegionSnapshot>> tasks = regions.stream()
                .map(region -> (Callable<RegionSnapshot>) () -> collectRegion(runAt, region)).toList();
        List<RegionSnapshot> parts = runParallel(tasks);

        List<CloudInstanceSnapshot> instanceRows = new ArrayList<>();
        List<CloudVpcSnapshot> vpcRows = new ArrayList<>();
        int failed = 0;
        for (RegionSnapshot part : parts) {
            if (part == null) {
                failed++;
                continue;
            }
            instanceRows.addAll(part.instances());
            vpcRows.addAll(part.vpcs());
        }
        // ??由ъ쟾 ?ㅽ뙣???ㅽ뙣???먭꺽利앸챸 ?먭린쨌?ㅽ듃?뚰겕 ?⑥젅 ?? ??SchedulerJob 洹쒖튃?濡??섏졇???대젰???④릿??
        if (!regions.isEmpty() && failed == regions.size()) {
            throw new IllegalStateException("cloud snapshot failed in all " + failed + " regions");
        }
        instanceRepository.saveAll(instanceRows);
        vpcRepository.saveAll(vpcRows);
        return String.format("regions=%d instances=%d vpcs=%d failed=%d",
                regions.size(), instanceRows.size(), vpcRows.size(), failed);
    }

    /** 蹂댁〈 湲곌컙 諛??ㅻ깄????젣. cleanup 諛곗튂媛 留ㅼ씪 ?몄텧?쒕떎. */
    @Transactional
    public String cleanup(int retentionDays) {
        LocalDateTime cutoff = LocalDateTime.now().minusDays(retentionDays);
        int instances = instanceRepository.deleteByRunAtBefore(cutoff);
        int vpcs = vpcRepository.deleteByRunAtBefore(cutoff);
        return String.format("deleted instances=%d vpcs=%d (retention=%dd)", instances, vpcs, retentionDays);
    }

    private RegionSnapshot collectRegion(LocalDateTime runAt, String region) {
        List<Instance> instances = aws.describeInstances(region);

        Map<String, String> statuses = new LinkedHashMap<>();
        List<String> ids = instances.stream().map(Instance::instanceId).toList();
        if (!ids.isEmpty()) {
            for (InstanceStatus s : aws.describeInstanceStatuses(region, ids)) {
                statuses.put(s.instanceId(), Ec2StatusRules.overallStatus(s));
            }
        }

        List<CloudInstanceSnapshot> instanceRows = new ArrayList<>();
        for (Instance i : instances) {
            String state = i.state() == null ? "unknown" : i.state().nameAsString();
            String check = statuses.get(i.instanceId());
            instanceRows.add(CloudInstanceSnapshot.builder()
                    .runAt(runAt).region(region)
                    .instanceId(i.instanceId()).name(nameTag(i.tags()))
                    .state(state).statusCheck(check)
                    .severity(Ec2StatusRules.classify(state, check))
                    .build());
        }

        List<CloudVpcSnapshot> vpcRows = new ArrayList<>();
        for (Vpc vpc : aws.describeVpcs(region)) {
            List<Subnet> subnets = aws.describeSubnetsByVpc(region, vpc.vpcId());
            List<RouteTable> routeTables = aws.describeRouteTablesByVpc(region, vpc.vpcId());
            List<SecurityGroup> groups = aws.describeSecurityGroupsByVpc(region, vpc.vpcId());
            List<Map<String, Object>> findings = NetworkExposureScanner.findings(subnets, routeTables, groups);
            long critical = findings.stream().filter(f -> "critical".equals(f.get("severity"))).count();
            // ?몄뒪?댁뒪 ?뚯냽? ?대? 諛쏆븘 ??由ъ쟾 紐⑸줉?먯꽌 ?몃㈃ ?쒕떎 ??describeInstancesByVpc ?ы샇異쒖? ??퉬.
            long inVpc = instances.stream().filter(i -> vpc.vpcId().equals(i.vpcId())).count();
            vpcRows.add(CloudVpcSnapshot.builder()
                    .runAt(runAt).region(region)
                    .vpcId(vpc.vpcId()).name(nameTag(vpc.tags()))
                    .subnetCount(subnets.size()).instanceCount((int) inVpc)
                    .criticalCount((int) critical).warningCount(findings.size() - (int) critical)
                    .findings(findings)
                    .build());
        }
        return new RegionSnapshot(instanceRows, vpcRows);
    }

    private String nameTag(List<Tag> tags) {
        return tags.stream().filter(t -> "Name".equals(t.key())).map(Tag::value).findFirst().orElse(null);
    }

    /** 由ъ쟾蹂?蹂묐젹 ?섏쭛. ?ㅽ뙣??由ъ쟾? null ???몄텧遺媛 failed濡??쇰떎. */
    private <T> List<T> runParallel(List<? extends Callable<T>> tasks) {
        if (tasks.isEmpty()) {
            return List.of();
        }
        ExecutorService pool = Executors.newFixedThreadPool(Math.min(tasks.size(), MAX_PARALLELISM));
        try {
            List<Future<T>> futures = tasks.stream().map(pool::submit).toList();
            List<T> results = new ArrayList<>(futures.size());
            for (Future<T> future : futures) {
                try {
                    results.add(future.get());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    results.add(null);
                } catch (Exception e) {
                    log.warn("cloud snapshot region task failed: {}", e.getMessage());
                    results.add(null);
                }
            }
            return results;
        } finally {
            pool.shutdown();
        }
    }

    private record RegionSnapshot(List<CloudInstanceSnapshot> instances, List<CloudVpcSnapshot> vpcs) {
    }
}

## 1. 소스 우선순위와 Q2

`zabbix-agent > cloudwatch > 밴드만`에는 동의합니다. 메모리·파일시스템 사용률은 기본 EC2 CloudWatch만으로 얻을 수 없고, 이미 보유한 Zabbix 데이터가 더 풍부하며 AWS 호출도 줄입니다.

다만 “agent에 매칭되면 CloudWatch를 절대 조회하지 않음”은 다음 경우에 관측 공백을 만들 수 있습니다.

- Zabbix agent는 연결돼 있지만 필요한 item(CPU/메모리/disk)이 없거나 최신 값이 오래된 경우
- Zabbix 장애·권한 문제·수집 지연이 발생한 경우
- CPU 의미가 서로 다른 경우(Zabbix guest/system CPU vs CloudWatch `CPUUtilization`)를 사후 비교해야 하는 경우

권장안은 평상시 병행 조회는 하지 않되, 매칭된 agent의 필수 지표와 freshness를 먼저 판정하는 것입니다. 모두 충족하면 CW를 끊고, Zabbix가 `SOURCE_UNAVAILABLE` 또는 stale일 때만 CW를 명시적 fallback으로 허용하는 편이 안전합니다. 이때 응답에는 `source`, `fallbackUsed`, `observedAt`를 남겨야 원인을 설명할 수 있습니다. 단순히 “Zabbix host와 매칭됨”은 충분한 조건이 아닙니다.

## 2. IP 매칭

“다중 매칭이면 연결하지 않음”은 반드시 필요한 안전장치지만, 단독 IP 일치만으로 오연결을 완전히 막지는 못합니다.

특히 보완이 필요합니다.

- EC2는 primary IP 외에 secondary private IP, 여러 ENI, ENI별 public/EIP를 가질 수 있습니다. 현재 상세 조립 로직은 `privateIp`·`publicIp` 중심이므로 ENI의 모든 주소를 후보에 넣어야 합니다.
- Zabbix interface는 agent뿐 아니라 SNMP/IPMI/JMX일 수 있습니다. `zabbix-agent` 소스로 채택하려면 적어도 `InterfaceKind.AGENT` 후보인지 확인해야 합니다.
- `useip=0`인 Zabbix host는 IP 대신 DNS로 접속할 수 있는데, 현재 `InventoryPort` 모델은 `ip`만 전달합니다. 이런 호스트는 매칭 누락으로 처리될 수 있습니다.
- 서로 다른 VPC·계정·온프레미스에 같은 RFC1918 IP가 존재할 수 있습니다. Zabbix proxy 뒤 호스트도 그 자체로 문제는 아니지만, 프록시 IP가 아니라 agent interface의 실제 대상 IP인지 확인해야 합니다.
- `Name` 태그 = Zabbix host name은 보조 신호로는 약합니다. 이름은 중복·변경이 흔합니다. instance ID 정확 일치나 Zabbix tag의 `aws.instanceId`, `aws.region`, `aws.accountId` 같은 명시적 식별자가 훨씬 안전합니다.

따라서 자동 연결 조건은 “양쪽 후보가 각각 하나”에 더해, 주소 정규화와 `AGENT` 인터페이스 제한을 적용하고, 매칭 근거를 `IP_EXACT`, `INSTANCE_ID_EXACT`, `NAME_EXACT`처럼 보존하는 것이 좋습니다. `NAME_EXACT`만으로 자동 연결하는 것은 권하지 않습니다.

## 3. 요청 시점 판정 vs 스냅샷 배치 저장

1차 구현은 **요청 시점 판정 + 5분 공유 캐시**가 더 적합합니다.

스냅샷 배치의 목적은 AWS 상태 이력 보존이고, Zabbix–EC2 연결은 바뀔 수 있는 운영 메타데이터입니다. 이를 스냅샷에 저장하면 과거 스냅샷의 연결 의미, 변경 시 재계산, 수동 연결 도입 시 정합성 문제가 생깁니다.

다만 `/ec2s` 목록의 커버리지 집계까지 고려하면, 단순한 “상세 요청마다 매칭”이 아니라 캐시에 다음을 보관하는 구조가 좋습니다.

- EC2 주소 → 후보 instance 목록
- Zabbix agent 주소 → 후보 host 목록
- 매칭 결과와 근거·판정 시각

중요한 빠진 점은 `InventoryPort`가 `CallerContext`로 Zabbix 세션을 해석한다는 것입니다. 사용자별 권한이나 토큰이 다른데 전역 캐시에 결과를 섞으면 권한 누출 또는 잘못된 결과가 날 수 있습니다. 캐시 키를 권한 범위별로 분리하거나, 이 기능 전용의 서비스 계정/시스템 컨텍스트를 명시적으로 설계해야 합니다. 배치에서 미리 계산하려면 후자가 사실상 필수입니다.

## 4. 커버리지와 P3 우선순위

개요 카드와 목록 컬럼을 함께 두는 판단에 동의합니다. 목록에 있어야 사용자가 “왜 이 인스턴스는 메모리·디스크가 없는가”를 즉시 이해하고 agent 설치 유도도 가능합니다.

표시값은 단순 세 종류보다 약간 더 구분하는 편이 좋습니다.

- `zabbix-agent`: 필수 지표가 있고 fresh함
- `cloudwatch`: CW 조회 가능
- `unavailable`: 권한 거부·예산 소진·수집 실패
- `unmonitored`: 어느 소스에도 실질 성능 지표가 없음

P3 `GetMetricData`의 우선순위 하향도 합리적입니다. agent 커버리지가 높으면 유료 fleet metric의 제품 가치는 낮아집니다. 다만 완전히 폐기하지 말고, 실제 “CW-only 인스턴스 수”, 상세 차트 요청량, fleet 비교 요구를 수집한 뒤 재평가하는 것이 좋습니다. P3의 가치는 단순 성능 차트가 아니라 다수 인스턴스 비교·집계 효율에 있습니다.

## 5. 추가 위험 및 단순화 제안

현재 `CloudMetricsService`는 CloudWatch 지표를 5개씩 요청하고, 접근 거부 캐시도 전역입니다. 새 설계에서는 최소한 접근 가능 여부와 매칭 캐시를 account/region 단위로 분리하는 편이 맞습니다. 한 리전의 권한 실패가 다른 리전의 정상 조회를 막으면 안 됩니다.

더 단순한 1차안은 다음입니다.

1. IP 정확 일치 + 유일 후보 + `AGENT` 인터페이스만 자동 매칭
2. Zabbix 필수 지표 존재·freshness 확인
3. 만족 시 agent 사용, 불만족 시 CW fallback
4. 이름 매칭·수동 연결·영속 `linked_hostid`는 2차
5. 목록/개요는 같은 매처 캐시를 사용

또한 source를 인스턴스 단위 하나로 고정하면 디스크·메모리만 agent, CPU·network만 CW인 상황을 표현하기 어렵습니다. 1차에서 단일 source 정책을 유지하더라도, API 내부 모델에는 지표별 source를 확장할 여지를 남기는 것이 좋습니다.

핵심 방향은 좋습니다. 다만 Q2는 “매칭되면 무조건 CW 중단”이 아니라 “agent 지표 품질이 확인되면 CW 중단”으로 바꾸고, IP 매칭에는 ENI 전체 주소·agent 인터페이스 제한·권한 안전 캐시를 추가해야 운영 중 관측 공백과 오연결을 피할 수 있습니다.

## 3. 요청 시점 판정 vs 스냅샷 배치 저장

1차 구현은 **요청 시점 판정 + 5분 공유 캐시**가 더 적합합니다.

스냅샷 배치의 목적은 AWS 상태 이력 보존이고, Zabbix–EC2 연결은 바뀔 수 있는 운영 메타데이터입니다. 이를 스냅샷에 저장하면 과거 스냅샷의 연결 의미, 변경 시 재계산, 수동 연결 도입 시 정합성 문제가 생깁니다.

다만 `/ec2s` 목록의 커버리지 집계까지 고려하면, 단순한 “상세 요청마다 매칭”이 아니라 캐시에 다음을 보관하는 구조가 좋습니다.

- EC2 주소 → 후보 instance 목록
- Zabbix agent 주소 → 후보 host 목록
- 매칭 결과와 근거·판정 시각

중요한 빠진 점은 `InventoryPort`가 `CallerContext`로 Zabbix 세션을 해석한다는 것입니다. 사용자별 권한이나 토큰이 다른데 전역 캐시에 결과를 섞으면 권한 누출 또는 잘못된 결과가 날 수 있습니다. 캐시 키를 권한 범위별로 분리하거나, 이 기능 전용의 서비스 계정/시스템 컨텍스트를 명시적으로 설계해야 합니다. 배치에서 미리 계산하려면 후자가 사실상 필수입니다.

## 4. 커버리지와 P3 우선순위

개요 카드와 목록 컬럼을 함께 두는 판단에 동의합니다. 목록에 있어야 사용자가 “왜 이 인스턴스는 메모리·디스크가 없는가”를 즉시 이해하고 agent 설치 유도도 가능합니다.

표시값은 단순 세 종류보다 약간 더 구분하는 편이 좋습니다.

- `zabbix-agent`: 필수 지표가 있고 fresh함
- `cloudwatch`: CW 조회 가능
- `unavailable`: 권한 거부·예산 소진·수집 실패
- `unmonitored`: 어느 소스에도 실질 성능 지표가 없음

P3 `GetMetricData`의 우선순위 하향도 합리적입니다. agent 커버리지가 높으면 유료 fleet metric의 제품 가치는 낮아집니다. 다만 완전히 폐기하지 말고, 실제 “CW-only 인스턴스 수”, 상세 차트 요청량, fleet 비교 요구를 수집한 뒤 재평가하는 것이 좋습니다. P3의 가치는 단순 성능 차트가 아니라 다수 인스턴스 비교·집계 효율에 있습니다.

## 5. 추가 위험 및 단순화 제안

현재 `CloudMetricsService`는 CloudWatch 지표를 5개씩 요청하고, 접근 거부 캐시도 전역입니다. 새 설계에서는 최소한 접근 가능 여부와 매칭 캐시를 account/region 단위로 분리하는 편이 맞습니다. 한 리전의 권한 실패가 다른 리전의 정상 조회를 막으면 안 됩니다.

더 단순한 1차안은 다음입니다.

1. IP 정확 일치 + 유일 후보 + `AGENT` 인터페이스만 자동 매칭
2. Zabbix 필수 지표 존재·freshness 확인
3. 만족 시 agent 사용, 불만족 시 CW fallback
4. 이름 매칭·수동 연결·영속 `linked_hostid`는 2차
5. 목록/개요는 같은 매처 캐시를 사용

또한 source를 인스턴스 단위 하나로 고정하면 디스크·메모리만 agent, CPU·network만 CW인 상황을 표현하기 어렵습니다. 1차에서 단일 source 정책을 유지하더라도, API 내부 모델에는 지표별 source를 확장할 여지를 남기는 것이 좋습니다.

## 총평

**보완 필요, 그러나 위 조건을 반영하면 진행 가능**입니다.

핵심 방향은 좋습니다. 다만 Q2는 “매칭되면 무조건 CW 중단”이 아니라 “agent 지표 품질이 확인되면 CW 중단”으로 바꾸고, IP 매칭에는 ENI 전체 주소·agent 인터페이스 제한·권한 안전 캐시를 추가해야 운영 중 관측 공백과 오연결을 피할 수 있습니다.
