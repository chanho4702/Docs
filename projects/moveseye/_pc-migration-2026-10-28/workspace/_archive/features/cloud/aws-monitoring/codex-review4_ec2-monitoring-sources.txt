    private <T> List<T> runParallel(List<? extends Callable<T>> tasks) {
        if (tasks.isEmpty()) {
            return List.of();
        }
        ExecutorService pool = Executors.newFixedThreadPool(Math.min(tasks.size(), MAX_PARALLELISM));
        try {
            List<Future<T>> futures = tasks.stream().map(pool::submit).toList();
            List<T> results = new ArrayList<>(futures.size());
            for (Future<T> future : futures) {
                try {
                    results.add(future.get());
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    results.add(null);
                } catch (Exception e) {
                    log.warn("cloud snapshot region task failed: {}", e.getMessage());
                    results.add(null);
                }
            }
            return results;
        } finally {
            pool.shutdown();
        }
    }
